Enter file contents here function calculateButtons(target) { const maxButtons = target + Math.max(...packages.map(p => p.buttons)) - 1; const dp = Array(maxButtons + 1).fill(null); dp[0] = { cost: 0, counts: [0, 0, 0, 0] }; for (let b = 0; b <= maxButtons; b++) { if (dp[b] === null) continue; packages.forEach((p, i) => { const nextButtons = b + p.buttons; if (nextButtons > maxButtons) return; const nextCost = dp[b].cost + p.price; if ( dp[nextButtons] === null || nextCost < dp[nextButtons].cost || ( nextCost === dp[nextButtons].cost && nextButtons > b ) ) { const counts = [...dp[b].counts]; counts[i]++; dp[nextButtons] = { cost: nextCost, counts: counts }; } }); } let best = null; for (let b = target; b <= maxButtons; b++) { if (dp[b] === null) continue; if ( best === null || dp[b].cost < best.cost || ( dp[b].cost === best.cost && b > best.buttons ) ) { best = { buttons: b, cost: dp[b].cost, counts: dp[b].counts }; } } if (best) { showResult( best.cost.toLocaleString("th-TH") + " บาท", "ได้รับ " + best.buttons.toLocaleString("th-TH") + " กระดุม", best.counts ); } }
